## Example Python code
import catboost
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error

# Assume 'data' is a pre-processed DataFrame
# CatBoost handles categorical features automatically. [23, 24, 25]
# Ensure categorical columns are of 'category' dtype
data['street_name'] = data['street_name'].astype('category')
data['violation_type'] = data['violation_type'].astype('category')

# Define features (X) and target (y)
features = ['street_name', 'violation_type', 'day_of_week', 'month'] # Add other engineered features
target = 'violation_count'

X = data[features]
y = data[target]

X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, shuffle=False)

# Identify categorical feature indices
cat_features_indices = [X.columns.get_loc(c) for c in ['street_name', 'violation_type']]

# Initialize and train the CatBoost model
cb_model = catboost.CatBoostRegressor(
    iterations=1000,
    learning_rate=0.05,
    depth=6,
    loss_function='RMSE',
    eval_metric='RMSE',
    random_seed=42,
    verbose=0,
    cat_features=cat_features_indices
)

cb_model.fit(
    X_train, y_train,
    eval_set=(X_test, y_test),
    early_stopping_rounds=50
)

# Predict and evaluate
predictions = cb_model.predict(X_test)
rmse = mean_squared_error(y_test, predictions, squared=False)
print(f"CatBoost RMSE: {rmse}")
